# IT-001 游戏状态同步集成测试

## 测试模块

| 项目 | 内容 |
|------|------|
| 测试ID | IT-001 |
| 模块 | 游戏状态同步 |
| 优先级 | P0 |
| 可并行 | ⚠️ 有限并行（最多2进程） |
| 依赖 | UT-001 ~ UT-009 |

---

## 1. 测试目标

验证多玩家通过WebSocket连接时，游戏状态能正确同步到所有客户端。

---

## 2. 测试场景

### 2.1 房间状态同步

| 用例ID | 场景 | 操作 | 预期同步结果 | 并行组 |
|--------|------|------|--------------|--------|
| IT-001-001 | 玩家加入 | P1加入房间 | 所有玩家收到join事件 | G1 |
| IT-001-002 | 玩家离开 | P1离开房间 | 所有玩家收到leave事件 | G1 |
| IT-001-003 | 玩家准备 | P1点击准备 | 所有玩家收到ready状态 | G1 |
| IT-001-004 | 房间满员 | 第4人加入 | 触发游戏开始检查 | G1 |

### 2.2 游戏状态同步

| 用例ID | 场景 | 操作 | 预期同步结果 | 并行组 |
|--------|------|------|--------------|--------|
| IT-001-010 | 发牌 | 游戏开始 | 每人收到自己的25张牌 | G2 |
| IT-001-011 | 叫主 | P1亮牌叫主 | 所有玩家收到叫主信息 | G2 |
| IT-001-012 | 出牌 | P1出牌 | 所有玩家收到出牌信息 | G2 |
| IT-001-013 | 得分 | 轮次结束 | 所有玩家收到更新后得分 | G2 |
| IT-001-014 | 回合结束 | 一轮结束 | 所有玩家收到轮次结果 | G2 |
| IT-001-015 | 游戏结束 | 牌出完 | 所有玩家收到最终结果 | G2 |

### 2.3 状态一致性

| 用例ID | 场景 | 验证点 | 并行组 |
|--------|------|--------|--------|
| IT-001-020 | 多客户端状态 | 所有客户端state一致 | G3 |
| IT-001-021 | 断线重连 | 重连后恢复正确状态 | G3 |
| IT-001-022 | 并发出牌 | 串行处理，无冲突 | G3 |

## 3. 测试代码

```javascript
const { Server } = require('socket.io');
const { createClient } = require('./test-utils');

describe('IT-001 游戏状态同步', () => {
  let io, clients;

  beforeAll(async () => {
    io = new Server(3001);
    await initGameServer(io);
  });

  afterAll(() => {
    io.close();
  });

  beforeEach(async () => {
    clients = await createClients(4);
  });

  afterEach(() => {
    clients.forEach(c => c.disconnect());
  });

  // G1: 房间状态同步
  describe('G1-房间状态', () => {
    it('IT-001-001 玩家加入应广播', async () => {
      const roomId = 'room-001';

      // P2, P3, P4 先监听
      const promises = clients.slice(1).map(c =>
        new Promise(resolve => c.on('room:playerJoined', resolve))
      );

      // P1 加入
      await clients[0].emit('room:join', { roomId });

      // 验证其他玩家收到通知
      const results = await Promise.all(promises);
      results.forEach(data => {
        expect(data.userId).toBe(clients[0].id);
      });
    });

    it('IT-001-003 玩家准备应同步', async () => {
      const roomId = 'room-001';
      await joinRoom(clients, roomId);

      const promises = clients.map(c =>
        new Promise(resolve => c.on('room:playerReady', resolve))
      );

      await clients[0].emit('room:ready', { isReady: true });

      const results = await Promise.all(promises);
      results.forEach(data => {
        expect(data.userId).toBe(clients[0].id);
        expect(data.isReady).toBe(true);
      });
    });
  });

  // G2: 游戏状态同步
  describe('G2-游戏状态', () => {
    it('IT-001-010 发牌应每人收到自己的牌', async () => {
      const roomId = 'room-001';
      await startGame(clients, roomId);

      const handPromises = clients.map(c =>
        new Promise(resolve => c.on('game:deal', resolve))
      );

      // 触发发牌
      await triggerDeal(roomId);

      const hands = await Promise.all(handPromises);
      hands.forEach(data => {
        expect(data.cards.length).toBe(25);
      });

      // 验证牌不重复
      const allCards = hands.flatMap(h => h.cards);
      const uniqueCards = new Set(allCards.map(c => c.id));
      expect(uniqueCards.size).toBe(allCards.length);
    });

    it('IT-001-012 出牌应广播给所有玩家', async () => {
      const roomId = 'room-001';
      const gameState = await startGame(clients, roomId);

      const playPromises = clients.map(c =>
        new Promise(resolve => c.on('game:playAnnounce', resolve))
      );

      // P1 出牌
      const cardToPlay = gameState.hands[0][0];
      await clients[0].emit('game:play', { cards: [cardToPlay] });

      const results = await Promise.all(playPromises);
      results.forEach(data => {
        expect(data.userId).toBe(clients[0].id);
        expect(data.cards).toEqual([cardToPlay]);
      });
    });
  });

  // G3: 状态一致性
  describe('G3-状态一致性', () => {
    it('IT-001-020 多客户端状态应一致', async () => {
      const roomId = 'room-001';
      await startGame(clients, roomId);

      // 进行若干轮出牌
      for (let round = 0; round < 5; round++) {
        await playRound(clients);
      }

      // 获取所有客户端状态
      const states = await Promise.all(
        clients.map(c => getState(c))
      );

      // 验证得分一致
      const scores = states.map(s => s.scores);
      expect(new Set(scores.map(JSON.stringify)).size).toBe(1);
    });
  });
});

// 测试工具函数
async function createClients(count) {
  return Promise.all(
    Array(count).fill(null).map(() => createClient('http://localhost:3001'))
  );
}

async function joinRoom(clients, roomId) {
  await Promise.all(
    clients.map(c => c.emit('room:join', { roomId }))
  );
}

async function startGame(clients, roomId) {
  await joinRoom(clients, roomId);
  await Promise.all(
    clients.map(c => c.emit('room:ready', { isReady: true }))
  );
  // 等待游戏开始
  return new Promise(resolve => {
    clients[0].on('game:start', resolve);
  });
}
```
