## 1. 测试目标

验证 `identifyCardType()` 函数能正确识别所有牌型。

## 4. 测试代码示例

```javascript
describe('UT-001 牌型识别', () => {

  // 并行组 G1: 单张
  describe.each([
    ['3', [{suit:'spade', rank:'3'}], 'single', '3'],
    ['A', [{suit:'heart', rank:'A'}], 'single', 'A'],
    ['大王', [{suit:'joker', rank:'big'}], 'single', 'big'],
    ['小王', [{suit:'joker', rank:'small'}], 'single', 'small'],
  ])('单张 - %s', (name, cards, expectedType, expectedRank) => {
    it(`应识别为${expectedType}`, () => {
      const result = identifyCardType(cards, '2');
      expect(result.valid).toBe(true);
      expect(result.type).toBe(expectedType);
      expect(result.rank).toBe(expectedRank);
    });
  });

  // 并行组 G2: 对子
  describe.each([
    ['2张', createCards('3', 2), 'pair', 2],
    ['3张', createCards('3', 3), 'pair', 3],
    ['4张', createCards('3', 4), 'pair', 4],
  ])('对子 - %s', (name, cards, expectedType, expectedCount) => {
    it(`应识别为${expectedType}`, () => {
      const result = identifyCardType(cards, '2');
      expect(result.valid).toBe(true);
      expect(result.type).toBe(expectedType);
      expect(result.cardCount).toBe(expectedCount);
    });
  });

  // 并行组 G3: 连对
  describe.each([
    ['3344', createConsecutivePairs(['3','4']), 2, true],
    ['334455', createConsecutivePairs(['3','4','5']), 3, true],
    ['2233打2时', createConsecutivePairs(['2','3']), 2, false],
    ['AAKK', createConsecutivePairs(['A','K']), 2, false],
  ])('连对 - %s', (name, cards, pairCount, expectedValid) => {
    it(`合法性应为${expectedValid}`, () => {
      const result = identifyCardType(cards, '2');
      if (expectedValid) {
        expect(result.valid).toBe(true);
        expect(result.type).toBe('consecutivePairs');
        expect(result.pairCount).toBe(pairCount);
      } else {
        expect(result.valid).toBe(false);
      }
    });
  });
});

// 辅助函数
function createCards(rank, count) {
  const suits = ['spade', 'heart', 'diamond', 'club'];
  return Array(count).fill(null).map((_, i) => ({
    suit: suits[i % 4],
    rank
  }));
}

function createConsecutivePairs(ranks) {
  return ranks.flatMap(rank => createCards(rank, 2));
}
```
