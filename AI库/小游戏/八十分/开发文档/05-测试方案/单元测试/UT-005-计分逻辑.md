## 1. 测试目标

## 2. 得分牌定义

| 牌 | 分值 |
|-----|------|
| K | 10分 |
| 10 | 10分 |
| 5 | 5分 |

## 4. 测试代码

```javascript
describe('UT-005 计分逻辑', () => {

  // G1: 单轮得分计算
  describe('G1-单轮得分', () => {
    it.each([
      // [牌, 预期得分]
      [['K'], 10],
      [['10'], 10],
      [['5'], 5],
      [['3'], 0],
      [['K', '10'], 20],
      [['K', 'K'], 20],
      [['K', '10', '5'], 25],
      [['K', 'K', '10', '10', '5', '5'], 50],
      [['3', '4', '6'], 0],
    ])('牌 %j 得分应为 %d', (cards, expected) => {
      expect(calculateRoundScore(cards)).toBe(expected);
    });
  });

  // G2: 多轮累计
  describe('G2-多轮累计', () => {
    it.each([
      [[10, 20, 15], 45],
      [[0, 0, 0], 0],
      [[25, 25, 25, 25], 100],
    ])('轮次得分 %j 累计应为 %d', (rounds, expected) => {
      const gameState = { scores: { team1: 0, team2: 0 } };
      rounds.forEach(score => {
        gameState.scores.team1 += score;
      });
      expect(gameState.scores.team1).toBe(expected);
    });
  });

  // G3: 队伍得分
  describe('G3-队伍得分', () => {
    it('守方赢轮应给守方加分', () => {
      const state = {
        defenderTeam: 1,
        scores: { team1: 0, team2: 0 }
      };
      const winnerId = 'player1'; // 假设player1在守方
      const scoreCards = createCards(['K']);

      updateTeamScore(state, winnerId, scoreCards);

      expect(state.scores.team1).toBe(10);
      expect(state.scores.team2).toBe(0);
    });
  });

  // G4: 边界值
  describe('G4-边界值', () => {
    it('空数组应返回0', () => {
      expect(calculateRoundScore([])).toBe(0);
    });

    it('所有得分牌总分应为100', () => {
      const allScoreCards = [
        ...createCards('K', 8),  // 4张×2副 = 8张
        ...createCards('10', 8),
        ...createCards('5', 8),
      ];
      expect(calculateRoundScore(allScoreCards)).toBe(100);
    });
  });
});

// 得分计算函数
function calculateRoundScore(cards) {
  const scoreMap = { 'K': 10, '10': 10, '5': 5 };
  return cards.reduce((sum, card) => sum + (scoreMap[card.rank] || 0), 0);
}
```
