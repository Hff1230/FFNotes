## 1. 测试目标

验证 `isValidPlay()` 函数能正确判断出牌是否合法。

## 3. 测试代码示例

```javascript
describe('UT-002 出牌合法性', () => {

  // 并行组 G1: 首出
  describe('G1-首出', () => {
    it.each([
      ['单张', ['♠3'], ['♠3'], true],
      ['对子', ['♠3','♠3'], ['♠3','♠3'], true],
      ['连对', ['♠3','♠3','♠4','♠4'], ['♠3','♠3','♠4','♠4'], true],
      ['非法牌型', ['♠3','♠4','♠5'], ['♠3','♠4','♠5'], false],
    ])('首出%s', (_, hand, play, expected) => {
      const result = isValidPlay(hand, play, null, '♥');
      expect(result.valid).toBe(expected);
    });
  });

  // 并行组 G2: 有同花色必须跟
  describe('G2-有同花色必须跟', () => {
    it.each([
      ['♠3♠5', '♠A', '♠3', true],
      ['♠3♠5', '♠A', '♥5', false],
      ['♠3♠3♥5', '♠A♠A', '♠3♠3', true],
      ['♠3♠3♥5', '♠A♠A', '♠3', false],
    ])('手牌[%s] 首出[%s] 出[%s]', (hand, lead, play, expected) => {
      const result = isValidPlay(
        parseCards(hand),
        parseCards(play),
        parseCards(lead),
        '♥'
      );
      expect(result.valid).toBe(expected);
    });
  });

  // 并行组 G3-G5 类似...
});
```
