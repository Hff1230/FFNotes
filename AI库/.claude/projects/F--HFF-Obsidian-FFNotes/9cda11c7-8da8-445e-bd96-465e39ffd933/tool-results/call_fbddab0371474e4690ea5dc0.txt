    79→// 创建一副麻将牌
    80→function createDeck() {
    81→    const deck = [];
    82→
    83→    // 筒条万 (每种4张)
    84→    for (const suit of SUITS) {
    85→        for (const rank of RANKS) {
    86→            for (let i = 0; i < 4; i++) {
    87→                deck.push({
    88→                    suit,
    89→                    rank,
    90→                    type: 'number',
    91→                    id: `${suit}${rank}_${i}`
    92→                });
    93→            }
    94→        }
    95→    }
    96→
    97→    // 风牌 (每种4张)
    98→    for (const wind of WINDS) {
    99→        for (let i = 0; i < 4; i++) {
   100→            deck.push({
   101→                suit: '风',
   102→                rank: wind,
   103→                type: 'wind',
   104→                id: `风${wind}_${i}`
   105→            });
   106→        }
   107→    }
   108→
   109→    // 箭牌 (每种4张)
   110→    for (const arrow of ARROWS) {
   111→        for (let i = 0; i < 4; i++) {
   112→            deck.push({
   113→                suit: '箭',
   114→                rank: arrow,
   115→                type: 'arrow',
   116→                id: `箭${arrow}_${i}`
   117→            });
   118→        }
   119→    }
   120→
   121→    return deck; // 136张
   122→}

   124→// 洗牌
   125→function shuffle(deck) {
   126→    const shuffled = [...deck];
   127→    for (let i = shuffled.length - 1; i > 0; i--) {
   128→        const j = Math.floor(Math.random() * (i + 1));
   129→        [shuffled[i], shuffled[j]] = [shuffled[j], shuffled[i]];
   130→    }
   131→    return shuffled;
   132→}

   197→        if (this.players.size >= this.maxPlayers) return false;

   207→            for (let i = 1; i <= 4; i++) {

   212→            }

   334→    // 获取第二次掷骰子的人（根据第一次骰子点数和，逆时针数）
   335→    getSecondRoller(firstSum) {
   336→        const playerIds = this.getSortedPlayerIds();
   337→        const firstRollerIndex = playerIds.indexOf(this.diceState.firstRoll.playerId);
   338→
   339→        // playerIds 是 [座位1, 座位2, 座位3, 座位4]（顺时针：东→南→西→北）
   340→        // 逆时针顺序：东(0)→北(3)→西(2)→南(1)→东(0)
   341→        // 逆时针移动一步 = 索引 -1（或 +3 mod 4）
   342→        // 数firstSum步，从1开始数（所以-1）
   343→        const secondRollerIndex = (firstRollerIndex - (firstSum - 1) % 4 + 4) % 4;
   344→        return playerIds[secondRollerIndex];
   345→    }

   370→
   371→            // 检查是否轮到AI掷第二次骰子

   374→            return { success: true, phase: 1, values };
   375→
   376→        } else if (this.diceState.rollPhase === 2) {
   377→            // 第二次掷骰子
   378→            this.diceState.secondRoll = { playerId, playerName: player.name, values };
   379→            this.diceState.rollPhase = 3;

   381→            this.broadcastMessage(`🎲 ${player.name} 掷出 ${values[0]}+${values[1]}=${values[0]+values[1]} 点`);

   396→        return { success: false, message: '骰子阶段错误' };

   399→    // 根据亮牌确定癞子（下一张牌）
   400→    getWildCardFromShown(shownCard) {
   401→        // 普通牌（筒条万）：1-9循环
   402→        if (shownCard.type === 'number') {
   403→            const nextRank = shownCard.rank >= 9 ? 1 : shownCard.rank + 1;
   404→            return {
   405→                suit: shownCard.suit,
   406→                rank: nextRank,
   407→                type: 'number'
   408→            };
   409→        }
   410→        // 风牌：东南西北循环
   411→        if (shownCard.type === 'wind') {
   412→            const windOrder = ['东', '南', '西', '北'];
   413→            const currentIndex = windOrder.indexOf(shownCard.rank);
   414→            const nextIndex = (currentIndex + 1) % 4;
   415→            return {
   416→                suit: '风',
   417→                rank: windOrder[nextIndex],
   418→                type: 'wind'
   419→            };
   420→        }
   421→        // 箭牌：中发白循环
   422→        if (shownCard.type === 'arrow') {
   423→            const arrowOrder = ['中', '发', '白'];
   424→            const currentIndex = arrowOrder.indexOf(shownCard.rank);
   425→            const nextIndex = (currentIndex + 1) % 3;
   426→            return {
   427→                suit: '箭',
   428→                rank: arrowOrder[nextIndex],
   429→                type: 'arrow'
   430→            };
   431→        }
   432→        return null;

   435→    // 确定万能牌（癞子）和抓牌位置 - 根据骰子结果
   436→    determineWildCard() {
   437→        const firstSum = this.diceState.firstRoll.values[0] + this.diceState.firstRoll.values[1];
   438→        const secondSum = this.diceState.secondRoll.values[0] + this.diceState.secondRoll.values[1];
   439→        const total = firstSum + secondSum;
   440→
   441→        console.log(`第1次骰子: ${firstSum}, 第2次骰子: ${secondSum}, 总计: ${total}`);
   442→
   443→        // 牌墙布局：每个玩家面前17墩牌（每墩2张），按逆时针排列
   444→        // 座位1(东) -> 座位4(北) -> 座位3(西) -> 座位2(南) -> 座位1(东)
   445→        // 牌墙索引：0-33(东), 34-67(北), 68-101(西), 102-135(南)
   446→        // 每个方位17墩，每墩2张牌
   447→
   448→        const firstRollerSeat = this.players.get(this.diceState.firstRoll.playerId)?.seatNum || 1;
   449→
   450→        // 根据第一次骰子确定从哪个方位开始数
   451→        // 逆时针方向：东(1) -> 北(4) -> 西(3) -> 南(2) -> 东(1)
   452→        const seatToDirection = { 1: '东', 2: '南', 3: '西', 4: '北' };
   453→        const directionOrder = [1, 4, 3, 2]; // 逆时针顺序
   454→
   455→        // 第1次骰子点数决定从哪个方位开始（从掷骰者的方位开始数）
   456→        const startDirectionIndex = directionOrder.indexOf(firstRollerSeat);
   457→        const countDirection = (startDirectionIndex + firstSum - 1) % 4;
   458→        const startDirection = directionOrder[countDirection];
   459→
   460→        console.log(`开始方位: ${seatToDirection[startDirection]} (座位${startDirection})`);
   461→
   462→        // 计算从开始方位最左边数起，数total墩
   463→        // 每个方位17墩
   464→        const pilesPerDirection = 17;
   465→        let remainingPiles = total;
   466→        let currentDirection = startDirection;
   467→        let directionIdx = directionOrder.indexOf(currentDirection);
   468→
   469→        // 数牌墩
   470→        while (remainingPiles > pilesPerDirection) {
   471→            remainingPiles -= pilesPerDirection;
   472→            directionIdx = (directionIdx + 1) % 4;
   473→            currentDirection = directionOrder[directionIdx];
   474→        }
   475→
   476→        // 确定抓牌起始位置
   477→        // 抓牌从第remainingPiles墩的下一墩开始
   478→        const drawPile = remainingPiles; // 第几墩（1-17）
   479→        const drawDirection = currentDirection;
   480→
   481→        console.log(`抓牌位置: ${seatToDirection[drawDirection]}方第${drawPile}墩后开始`);
   482→
   483→        // 确定亮牌位置
   484→        // 抓牌那墩的逆时针玩家面前的牌墩从右往左数第一次骰子的数量
   485→        const drawDirectionIdx = directionOrder.indexOf(drawDirection);
   486→        const shownDirectionIdx = (drawDirectionIdx + 1) % 4; // 逆时针下一个
   487→        const shownDirection = directionOrder[shownDirectionIdx];
   488→
   489→        // 从右往左数第firstSum墩
   490→        const shownPile = pilesPerDirection - firstSum + 1; // 从右往左数
   491→
   492→        console.log(`亮牌位置: ${seatToDirection[shownDirection]}方第${shownPile}墩（从右往左数第${firstSum}墩）`);
   493→
   494→        // 计算牌堆中的实际索引
   495→        // 牌墙按 [东17墩][北17墩][西17墩][南17墩] 排列
   496→        const directionToOffset = { 1: 0, 4: 17, 3: 34, 2: 51 }; // 座位到牌墩偏移
   497→        const shownPileIndex = directionToOffset[shownDirection] + (shownPile - 1);
   498→
   499→        // 每墩2张牌，取上面那张（索引*2）
   500→        const shownCardIndex = shownPileIndex * 2;
   501→
   502→        // 获取亮牌
   503→        let shownCard = null;
   504→        if (shownCardIndex >= 0 && shownCardIndex < this.gameState.deck.length) {
   505→            shownCard = this.gameState.deck[shownCardIndex];
   506→            console.log(`亮牌索引: ${shownCardIndex}, 亮牌: ${cardName(shownCard)}`);
   507→

   591→        // 逆时针需要调整为 [座位1, 座位4, 座位3, 座位2]
   592→        const counterClockwiseIds = [
   593→            playerIds[0], // 座位1(东)

   765→    // 排序手牌
   766→    sortHand(playerId) {
   767→        const hand = this.gameState.hands[playerId];
   768→        if (!hand) return;
   769→
   770→        const suitOrder = { '筒': 0, '条': 1, '万': 2, '风': 3, '箭': 4 };
   771→        const windOrder = { '东': 0, '南': 1, '西': 2, '北': 3 };
   772→        const arrowOrder = { '中': 0, '发': 1, '白': 2 };
   773→
   774→        hand.sort((a, b) => {
   775→            // 万能牌放最后
   776→            if (this.isWildCard(a) && !this.isWildCard(b)) return 1;
   777→            if (!this.isWildCard(a) && this.isWildCard(b)) return -1;
   778→
   779→            const suitA = suitOrder[a.suit] ?? 5;
   780→            const suitB = suitOrder[b.suit] ?? 5;
   781→            if (suitA !== suitB) return suitA - suitB;
   782→
   783→            if (a.type === 'number' && b.type === 'number') {
   784→                return a.rank - b.rank;
   785→            } else if (a.type === 'wind' && b.type === 'wind') {
   786→                return (windOrder[a.rank] ?? 0) - (windOrder[b.rank] ?? 0);
   787→            } else if (a.type === 'arrow' && b.type === 'arrow') {
   788→                return (arrowOrder[a.rank] ?? 0) - (arrowOrder[b.rank] ?? 0);
   789→            }
   790→            return 0;
   791→        });
   792→    }

   804→        return false;

  1021→    // 检查并执行AI行动
